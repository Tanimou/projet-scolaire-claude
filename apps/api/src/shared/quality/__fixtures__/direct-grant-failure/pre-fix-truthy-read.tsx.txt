export function PreFixSecurityPanel({ me }: { me: MeResponse | null }) {
  return (
    <div>
      {me.mfaEnabled && <Badge>MFA actif</Badge>}
      <Field label="Double authentification" value={me?.mfaEnabled ? 'Activée' : 'Recommandée'} />
      {!me.mfaEnabled ? <Hint>Configurez votre MFA</Hint> : null}
    </div>
  );
}
