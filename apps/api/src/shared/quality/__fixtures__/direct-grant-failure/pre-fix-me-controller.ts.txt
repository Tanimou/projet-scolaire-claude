export class PreFixMeController {
  async me(jwt: KeycloakJwtPayload) {
    const user = await this.users.ensureUser(jwt);
    const realmRoles = jwt.realm_access?.roles ?? [];
    const payload = {
      id: user.id,
      email: user.email,
      roles: realmRoles,
      schoolId: null,
