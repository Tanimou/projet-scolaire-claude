const MFA_ROLES = ['school_admin', 'teacher'];

export function preFixInviteRequiredActions(body: { realmRole: string }) {
  const requiredActions = ['UPDATE_PASSWORD'];
  if (body.realmRole === 'school_admin' || body.realmRole === 'teacher') {
    requiredActions.push('CONFIGURE_TOTP');
  }
  return { requiredActions, MFA_ROLES };
}
