/**

 * en « ${n} prochains ».

  'new Date(2026, 9, 1, 23, 59, 59)',
  'Date.now()',
  'events.filter((e) => e.endsAt >= now).slice(0, 12)',
  'today.getMonth() === month.getMonth()',
